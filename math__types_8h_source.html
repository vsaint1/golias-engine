253  class Random {
254 
255  public:
256  // Random in [min, max] for integers, [min, max) for floats
257  template <typename T, typename = std::enable_if_t<std::is_arithmetic_v<T>>>
258  static T Range(T min, T max) {
259  if constexpr (std::is_integral_v<T>) {
260  std::uniform_int_distribution<T> dist(min, max);
261  return dist(GetEngine());
262  } else if constexpr (std::is_floating_point_v<T>) {
263  std::uniform_real_distribution<T> dist(min, max);
264  return dist(GetEngine());
265  }
266  }
267 
268  // Random in [0, 1] for floats, [0, MAX] for integers, true/false for bool
269  template <typename T, typename = std::enable_if_t<std::is_arithmetic_v<T> || std::is_same_v<T, bool>>>
270  static T Value() {
271  if constexpr (std::is_same_v<T, bool>) {
272  return Range<int>(0, 1) == 1;
273  } else if constexpr (std::is_floating_point_v<T>) {
274  return Range<T>(T(0), T(1));
275  } else if constexpr (std::is_integral_v<T>) {
276  std::bernoulli_distribution dist(0.5);
277  return dist(GetEngine());
278  }
279  }
280 
281  private:
282  static std::mt19937& GetEngine();
283  };
