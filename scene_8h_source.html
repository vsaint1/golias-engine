124  template <typename T>
125  void RemoveSystem() {
126  auto it = mSystemIndex.find(std::type_index(typeid(T)));
127  if (it == mSystemIndex.end()) {
128  return;
129  }
130  mSystems[it->second]->OnDestroy(*this);
131  mSystems.erase(mSystems.begin() + static_cast<ptrdiff_t>(it->second));
132  // Rebuild index after erasure
133  mSystemIndex.clear();
134  for (size_t i = 0; i < mSystems.size(); ++i) {
135  mSystemIndex[std::type_index(typeid(*mSystems[i]))] = i;
136  }
